import { reconcilePids, ReconciliationReport } from '../pidReconciler.js';

enum ExitCode {
  SUCCESS = 0,
  VALIDATION_FAILED = 1,
  RUNTIME_ERROR = 2,
}

type LogLevel = 'info' | 'error' | 'warn' | 'debug';

interface LogEntry {
  timestamp: string;
  level: LogLevel;
  message: string;
  payload?: unknown;
}

interface ErrorNormalized {
  name: string;
  message: string;
  stack?: string;
}

const emit = (level: LogLevel, message: string, payload?: unknown): void => {
  const entry: LogEntry = {
    timestamp: new Date().toISOString(),
    level,
    message,
    ...(payload !== undefined && { payload }),
  };

  const output = JSON.stringify(entry);

  if (level === 'error') {
    console.error(output);
  } else {
    console.log(output);
  }
};

const normalizeError = (error: unknown): ErrorNormalized => {
  const err = error as Error;
  return {
    name: err?.name || 'Error',
    message: err?.message || 'Unknown error',
    stack: err?.stack,
  };
};

const setExitCode = (code: ExitCode): void => {
  process.exitCode = code;
};

async function runReconciliation(): Promise<ReconciliationReport> {
  emit('info', 'PID reconciliation started');

  try {
    const report = await reconcilePids(config);
    
    emit('info', 'PID reconciliation completed', report);
    
    setExitCode(report.passed ? ExitCode.SUCCESS : ExitCode.VALIDATION_FAILED);
    
    return report;
  } catch (error) {
    emit(
      'error',
      'PID reconciliation job failed during execution',
      normalizeError(error)
    );
    
    setExitCode(ExitCode.RUNTIME_ERROR);
    
    throw error;
  }
}

async function main(): Promise<void> {
  try {
    await runReconciliation();
  } catch (error) {
    // Error ya fue emitido en runReconciliation, pero mantenemos este bloque
    // para capturar cualquier error inesperado en la llamada misma
    if (process.exitCode === undefined) {
      emit('error', 'Unexpected error in main', normalizeError(error));
      setExitCode(ExitCode.RUNTIME_ERROR);
    }
  }
}

main().finally(() => {

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import torch
from loguru import logger
from typing import Optional
import time

app = FastAPI(title="Isabella AI - Inference")

class InferenceRequest(BaseModel):
text: str
max_tokens: Optional[int] = 128
temperature: Optional[float] = 0.7

Cargar modelo globalmente (singleton)
MODEL = None
TOKENIZER = None
DEVICE = "cuda" if torch.cuda.is_available() else "cpu"

def load_model():
global MODEL, TOKENIZER
if MODEL is None:
from transformers import AutoModelForCausalLM, AutoTokenizer
MODEL_NAME = "path_or_hub_model"
TOKENIZER = AutoTokenizer.from_pretrained(MODEL_NAME)
MODEL = AutoModelForCausalLM.from_pretrained(MODEL_NAME).to(DEVICE)
return MODEL, TOKENIZER

@app.on_event("startup")
def startup_event():
start = time.time()
load_model()
logger.info("Modelo cargado en {:.2f}s".format(time.time() - start))

@app.get("/health")
def health():
return {"status": "ok", "device": DEVICE}

@app.post("/infer")
def infer(req: InferenceRequest):
model, tokenizer = load_model()

inputs = tokenizer(req.text, return_tensors="pt").to(DEVICE)
with torch.no_grad():
out = model.generate(**inputs, max_new_tokens=req.max_tokens, do_sample=True, temperature=req.temperature)
text = tokenizer.decode(out, skip_special_tokens=True)
return {"text": text}
except Exception as e:
logger.exception("Error en inferencia")
raise HTTPException(status_code=500, detail=str(e))

return yaml.safe_load(f)

def main():

ds = load_dataset(cfg["dataset"]["name"])
tokenizer = AutoTokenizer.from_pretrained(cfg["model"]["base"])
def tokenize_fn(ex):
return tokenizer(ex["text"], truncation=True, max_length=cfg["dataset"]["max_length"])
tokenized = ds.map(tokenize_fn, batched=True)
model = AutoModelForCausalLM.from_pretrained(cfg["model"]["base"])
training_args = TrainingArguments(**cfg["training"])
trainer = Trainer(model=model, args=training_args, train_dataset=tokenized["train"])
trainer.train()
trainer.save_model(cfg["output"]["dir"])

if _name_ == "_main_":
main()

from transformers import AutoModelForCausalLM, AutoTokenizer
from peft import PeftModel, PeftConfig
import torch
import os

class IsabellaModel:
    def __init__(
        self,
        model_name_or_path: str,
        device: str = None,
        lora_path: str = None
    ):
        self.device = device or ("cuda" if torch.cuda.is_available() else "cpu")
        self.tokenizer = AutoTokenizer.from_pretrained(
            model_name_or_path,
            use_fast=True,
            trust_remote_code=True
        )
        
        # Cargar modelo base
        self.base_model = AutoModelForCausalLM.from_pretrained(
            model_name_or_path,
            torch_dtype=torch.float16 if self.device.startswith("cuda") else torch.float32,
            trust_remote_code=True
        )
        
        # Aplicar LoRA si existe
        if lora_path and os.path.exists(lora_path):
            self.model = PeftModel.from_pretrained(self.base_model, lora_path)
        else:
            self.model = self.base_model
            
        self.model.to(self.device)
        self.model.eval()

    def generate(
        self,
        prompt: str,
        max_new_tokens: int = 256,
        temperature: float = 0.7,
        **kwargs
    ):
        inputs = self.tokenizer(prompt, return_tensors="pt").to(self.device)
        with torch.no_grad():
            out = self.model.generate(
                **inputs,
                max_new_tokens=max_new_tokens,
                do_sample=True if temperature > 0 else False,
                temperature=temperature,
                **kwargs
            )
        return self.tokenizer.decode(out[0], skip_special_tokens=True)

    def load_lora_adapter(self, lora_path: str):
        """Cargar adapter LoRA en caliente"""
        self.model = PeftModel.from_pretrained(self.base_model, lora_path)

from sentence_transformers import SentenceTransformer
import faiss
import numpy as np
import pickle
import os

class Retriever:
    def __init__(
        self,
        embed_model_name: str = "all-MiniLM-L6-v2",
        index_path: str = None
    ):
        self.embedder = SentenceTransformer(embed_model_name)
        self.index = None
        self.metadata = []
        self.index_path = index_path
        
        if index_path and os.path.exists(index_path):
            try:
                with open(index_path, "rb") as f:
                    self.index, self.metadata = pickle.load(f)
            except Exception as e:
                print(f"Error loading index: {e}")
                self.index = None

    def build_index(self, texts: list):
        """Construir índice FAISS desde lista de textos"""
        embeddings = self.embedder.encode(texts, convert_to_numpy=True, normalize_embeddings=True)
        dim = embeddings.shape[1]
        self.index = faiss.IndexFlatIP(dim)
        self.index.add(embeddings)
        self.metadata = texts

    def save_index(self, path: str):
        """Guardar índice en disco"""
        with open(path, "wb") as f:
            pickle.dump((self.index, self.metadata), f)

    def retrieve(self, query: str, top_k: int = 5):
        """Recuperar top_k documentos más relevantes"""
        if not self.index:
            return []
        q_emb = self.embedder.encode([query], convert_to_numpy=True, normalize_embeddings=True)
        D, I = self.index.search(q_emb, top_k)
        return [self.metadata[i] for i in I[0] if i < len(self.metadata)]

    def add_documents(self, new_texts: list):
        """Añadir documentos al índice existente"""
        if not self.index:
            self.build_index(new_texts)
            return
        new_embeddings = self.embedder.encode(new_texts, convert_to_numpy=True, normalize_embeddings=True)
        self.index.add(new_embeddings)
        self.metadata.extend(new_texts)
